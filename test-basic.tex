%% test-basic.tex
%% ==============
%% Typesetting test for every engine: breakable boxes inside breakable
%% boxes.
%%
%%   pdflatex test-basic.tex   (or xelatex / lualatex)
%%   platex  test-basic.tex && dvipdfmx test-basic
%%   uplatex -jobname=test-basic "\def\CLS{ujarticle}\input test-basic"
%% The class is article by default; set \CLS for another one.  With
%% pTeX the class option dvipdfmx is given.
%%
%% What is covered:
%%   * a breakable box inside a breakable box, broken across several
%%     pages, with a list and a display inside;
%%   * a breakable box inside this one, also broken, with an ordinary
%%     box inside;
%%   * a small box with a title, a box of the enhanced skin;
%%   * a box that is the first thing in the outer box and has a text
%%     color of its own, a box right after it, a box in an item of a
%%     list, a box defined by \newtcolorbox;
%%   * a box in the lower part of the outer box;
%%   * a breakable box in a minipage and in an unbreakable box: these
%%     do not flow;
%%   * boxes in an outer box that is not broken, one with a lower part.
%%
%% What to check:
%%   * TNB-RESULT OK, and no error, warning or overfull box;
%%   * in the PDF, the frames are behind their text and open where the
%%     outer box is broken.
\ifdefined\CLS\else\def\CLS{article}\fi
\ifdefined\ptexversion\def\CLSOPT{dvipdfmx}\else\def\CLSOPT{}\fi
\expandafter\documentclass\expandafter[\CLSOPT]{\CLS}
\usepackage{tcbnestbreak}
\input{test-common}
\newtcolorbox{note}[2][]{breakable, colback=orange!10,
  colframe=orange!70!black, title={#2}, #1}
\begin{document}
\section*{Breakable boxes inside breakable boxes}

\TNBfiller{20}

\begin{tcolorbox}[breakable, title=Outer box 1]

\begin{tcolorbox}[breakable, colback=yellow!20, colframe=blue!60!black,

\item Another item with a display
  \[ a^2 + b^2 = c^2 \]
  and text after it.
\end{itemize}
\begin{tcolorbox}[breakable, colback=red!10, colframe=red!60!black,
  title=Box 2 in box 1]

\TNBfiller{4}
\end{tcolorbox}
\TNBfiller{60}
End of box 2.
\end{tcolorbox}
\TNBfiller{30}
End of box 1.
\end{tcolorbox}
\TNBfiller{8}
\begin{tcolorbox}[breakable, colback=green!10, title=Box 3]
A small box.
\end{tcolorbox}
\TNBfiller{30}

  colframe=cyan!50!black]
Start of box 4, of the enhanced skin. \TNBfiller{120}
End of box 4.
\end{tcolorbox}
\TNBfiller{8}
\end{tcolorbox}

\begin{tcolorbox}[breakable, title=Outer box 2]

\TNBfiller{8}
\end{note}
\begin{enumerate}

\end{note}
\item The second item.
\end{enumerate}

\end{minipage}

The lower part of the outer box. \TNBfiller{8}
\begin{note}{Box 8, in the lower part}
\TNBfiller{120}
End of box 8.
\end{note}
\TNBfiller{8}
\end{tcolorbox}
